// =====================================================================
// DRAWING X-COORDINATE GUARD
// TradingView rejects drawings whose bar_index x-coordinate is too far

    na(x) ? x : math.max(x, bar_index - 4999)

// =====================================================================
// BASIC CANDLE METRICS
// =====================================================================
c_body  = math.abs(close - open)
c_range = math.max(high - low, syminfo.mintick)
bodyPct = c_body / c_range

// Half a tick: tolerance for ratio tests on tick-rounded prices. Without it a
// candle that is EXACTLY 30% / 70% of another can fail by floating-point noise
// (e.g. 105.1 - 101.9 = 3.19999...).
float eps = syminfo.mintick * 0.5

upper_wick = high - math.max(close, open)
lower_wick = math.min(close, open) - low

is_bull       = close > open
is_bear       = close < open

atr           = ta.atr(14)

// =====================================================================
// CANDLE CLASSIFIER (V18.0)
// Every test is a ratio of a candle part to the candle's own range
// (High - Low), never an absolute price size. Checked in priority order
// Doji -> Pin Bar -> Marubozu -> Normal: a candle that fits several types
// takes the first one. A candle with High == Low has no type.
// =====================================================================
bool  cdlValid   = high > low
float cdlRange   = high - low
float cdlBodyR   = cdlValid ? c_body / cdlRange : na
float cdlUpR     = cdlValid ? upper_wick / cdlRange : na
float cdlLoR     = cdlValid ? lower_wick / cdlRange : na
float cdlOpenPos = cdlValid ? (open - low) / cdlRange : na
float cdlClosePos = cdlValid ? (close - low) / cdlRange : na

// Doji: small body, a noticeable shadow on BOTH sides, Open and Close near the centre
bool is_doji = cdlValid and cdlBodyR <= cdlDojiMaxBody and cdlUpR >= cdlDojiMinShadow and cdlLoR >= cdlDojiMinShadow and math.abs(cdlOpenPos - 0.5) <= cdlDojiCenterTol and math.abs(cdlClosePos - 0.5) <= cdlDojiCenterTol

// Pin Bar: direction comes from the DOMINANT shadow, not from the candle colour
//   Bullish Pin: long lower shadow; Bearish Pin: long upper shadow
bool cdlBullPinRatio = c_body > 0 ? lower_wick >= c_body * cdlPinShadowBody : lower_wick > 0
bool cdlBearPinRatio = c_body > 0 ? upper_wick >= c_body * cdlPinShadowBody : upper_wick > 0
bool is_bull_pin = cdlValid and not is_doji and cdlBodyR <= cdlPinMaxBody and cdlLoR >= cdlPinMinShadow and cdlBullPinRatio and cdlUpR <= cdlPinMaxOpp
bool is_bear_pin = cdlValid and not is_doji and cdlBodyR <= cdlPinMaxBody and cdlUpR >= cdlPinMinShadow and cdlBearPinRatio and cdlLoR <= cdlPinMaxOpp
bool is_pinbar   = is_bull_pin or is_bear_pin

// Marubozu: very large body, small total shadows
bool is_marubozu = cdlValid and not is_doji and not is_pinbar and c_body >= bo_maru_body * cdlRange - eps and upper_wick + lower_wick <= cdlMaruMaxShadow * cdlRange + eps

// Normal: medium body (min <= body < max)
bool is_normal = cdlValid and not is_doji and not is_pinbar and not is_marubozu and cdlBodyR >= cdlNormMinBody and cdlBodyR < cdlNormMaxBody

barcolor(cdlColorBars ? (is_doji ? color.yellow : is_bull_pin ? color.lime : is_bear_pin ? color.red : is_marubozu and is_bull ? color.green : is_marubozu and is_bear ? color.maroon : is_normal and is_bull ? color.aqua : is_normal and is_bear ? color.orange : na) : na)

// =====================================================================
// BIG MARUBOZU / BIG PIN BAR TRACKERS (Pure Prior 5 History)
// Big Marubozu: its length (High-Low) is larger than at least 3 of the 5
//   preceding Marubozus of the SAME colour. With fewer than 3 same-colour
//   Marubozus, the last 5 Marubozus of ANY colour are used. No previous
//   Marubozu at all -> not Big.
// Big Pin Bar: same logic with the 5 preceding Pin Bars of the SAME direction
//   (fallback: the last 5 Pin Bars of any direction).
// =====================================================================
var float[] maru_bull_sizes = array.new_float(0)
var float[] maru_bear_sizes = array.new_float(0)
var float[] maru_all_sizes  = array.new_float(0)
var float[] pin_bull_sizes  = array.new_float(0)
var float[] pin_bear_sizes  = array.new_float(0)
var float[] pin_all_sizes   = array.new_float(0)

f_bigger_count(float[] sizes, float len) =>
    int cnt = 0
    if array.size(sizes) > 0
        for i = 0 to array.size(sizes) - 1
            if len > array.get(sizes, i)
                cnt += 1
    cnt

f_push5(float[] sizes, float len) =>
    array.push(sizes, len)
    if array.size(sizes) > 5
        array.shift(sizes)
    true

int bigCount = 0
if is_bull
    if array.size(maru_bull_sizes) >= 3
        bigCount := f_bigger_count(maru_bull_sizes, c_range)
    else
        bigCount := f_bigger_count(maru_all_sizes, c_range)
else if is_bear
    if array.size(maru_bear_sizes) >= 3
        bigCount := f_bigger_count(maru_bear_sizes, c_range)
    else
        bigCount := f_bigger_count(maru_all_sizes, c_range)
bool is_big_maru = is_marubozu and bigCount >= 3

int pinCount = 0
if is_bull_pin
    if array.size(pin_bull_sizes) >= 3
        pinCount := f_bigger_count(pin_bull_sizes, c_range)
    else
        pinCount := f_bigger_count(pin_all_sizes, c_range)
else if is_bear_pin
    if array.size(pin_bear_sizes) >= 3
        pinCount := f_bigger_count(pin_bear_sizes, c_range)
    else
        pinCount := f_bigger_count(pin_all_sizes, c_range)
bool is_big_pin = is_pinbar and pinCount >= 3

if is_marubozu
    f_push5(maru_all_sizes, c_range)
    if is_bull
        f_push5(maru_bull_sizes, c_range)
    if is_bear
        f_push5(maru_bear_sizes, c_range)
if is_pinbar
    f_push5(pin_all_sizes, c_range)
    if is_bull_pin
        f_push5(pin_bull_sizes, c_range)
    if is_bear_pin
        f_push5(pin_bear_sizes, c_range)

// =====================================================================
// SPECIAL THREE-CANDLE PATTERNS (Price action confirmation)
// C1 = bar[2], C2 = bar[1], C3 = current bar; candles in the direction of the
// move. Coloured candles (Marubozu, Normal) must have the move's colour; a Pin
// Bar must point in the move's direction (its colour does not matter); a Doji
// has no colour condition.
//  A: Big Marubozu + Pin Bar or Doji + Big Marubozu
//  B: Pin Bar or Doji + Marubozu or Normal + Big Marubozu; C2 is an inside bar
//     of C1, or closes at most 5% of C1's length short of C1's far end, or beyond
//  C: Normal + Normal (each at least 40% of C3's length) + Big Marubozu
// C3 closes beyond the extreme (high / low) of C1 and C2.
// They confirm a Real Breakout (with the level test in f_tri_at) and a
// Real Pullback (no level).
// =====================================================================
bool tri_in = high[1] <= high[2] and low[1] >= low[2]
// Pattern C: each Normal at least tri_c_norm_pct (40%) of C3's length
bool tri_nC = c_range[2] >= tri_c_norm_pct * c_range - eps and c_range[1] >= tri_c_norm_pct * c_range - eps

// Bullish
bool tri_c3_up   = close > math.max(high[1], high[2])
bool tri_near_up = close[1] >= high[2] - tri_near_pct * c_range[2] - eps
bool triA_up = useTri and is_big_maru[2] and is_bull[2] and (is_bull_pin[1] or is_doji[1]) and is_big_maru and is_bull and tri_c3_up
bool triB_up = useTri and (is_bull_pin[2] or is_doji[2]) and (is_marubozu[1] or is_normal[1]) and is_bull[1] and (tri_in or tri_near_up) and is_big_maru and is_bull and tri_c3_up
bool triC_up = useTri and is_normal[2] and is_bull[2] and is_normal[1] and is_bull[1] and tri_nC and is_big_maru and is_bull and tri_c3_up

// Bearish
bool tri_c3_dn   = close < math.min(low[1], low[2])
bool tri_near_dn = close[1] <= low[2] + tri_near_pct * c_range[2] + eps
bool triA_dn = useTri and is_big_maru[2] and is_bear[2] and (is_bear_pin[1] or is_doji[1]) and is_big_maru and is_bear and tri_c3_dn
bool triB_dn = useTri and (is_bear_pin[2] or is_doji[2]) and (is_marubozu[1] or is_normal[1]) and is_bear[1] and (tri_in or tri_near_dn) and is_big_maru and is_bear and tri_c3_dn
bool triC_dn = useTri and is_normal[2] and is_bear[2] and is_normal[1] and is_bear[1] and tri_nC and is_big_maru and is_bear and tri_c3_dn

bool tri_up = triA_up or triB_up or triC_up
bool tri_dn = triA_dn or triB_dn or triC_dn

// Level test for the patterns (Real Breakout / range exit):
//  A: C1 AND C3 close beyond the level; B and C: C3 closes beyond the level.
f_tri_at(level, dir) =>
    bool ok = false
    if not na(level)
        if dir == 1
            ok := (triA_up and close[2] > level and close > level) or ((triB_up or triC_up) and close > level)
        else if dir == -1
            ok := (triA_dn and close[2] < level and close < level) or ((triB_dn or triC_dn) and close < level)
    ok

// Low opposite pressure on C2 (Rule 2 and 3 of the Real Breakout AND the Real Pullback):
//  C2 of an UP move is not a Bearish Pin Bar (long upper shadow, body low);
//  C2 of a DOWN move is not a Bullish Pin Bar (long lower shadow, body high).
bool  c2_low_sell = not is_bear_pin   // C2 of an up move
bool  c2_low_buy  = not is_bull_pin   // C2 of a down move

// =====================================================================
// REAL BREAKOUT (3 Main Conditions) - used by market structure AND the Range Box
// A breakout of `level` (the BOI line) is REAL only when it matches one of the
// three main conditions below. Anything else that crosses the level is a
// Fakeout. Every rule is a closed two-candle pattern (C1 = previous candle,
// C2 = current candle) evaluated on the bar it completes.
//
// Rule 1: C1 and C2 are Marubozus in the breakout direction, both close beyond
//         the BOI, C2 closes beyond the ENTIRE C1 candle (its high in an
//         up-breakout, its low in a down-breakout) and C2's length (High-Low)
//         is at least 70% of C1's length. No % beyond BOI required.
// Rule 2: C1 is a BIG Marubozu with MORE than 30% of its whole candle beyond
//         the BOI; C2 is a candle in the breakout direction (any size) whose
//         low (high for down-breakouts) stays beyond the 50% mark of C1, on
//         low volume.
// Rule 3: as Rule 2, but C2 is opposite to the breakout direction and at most
//         30% of C1's length (High-Low).
// =====================================================================
f_real_breakout(level, dir) =>
    bool ok = false
    if not na(level)
        if dir == 1
            // Rule 1: two bullish Marubozus, both closing above the BOI, C2 closing above C1's high
            bool r1 = is_marubozu[1] and is_marubozu and is_bull[1] and is_bull and (close[1] > level) and (close > level) and (close > high[1]) and (c_range >= bo_c2_min_r1 * c_range[1] - eps)
            // More than 30% of C1's WHOLE candle (body + shadows) above the CTS
            float c1_part_above = math.max(0.0, high[1] - math.max(low[1], level))
            bool c1_above_30 = c1_part_above > bo_c1_above_pct * c_range[1] + eps

            float mid_c1 = (high[1] + low[1]) / 2.0
            // Rule 2: big bullish Marubozu + bullish candle (low volume) whose LOW stays above C1's 50%
            bool r2 = is_big_maru[1] and is_bull[1] and is_bull and (low > mid_c1) and c2_low_sell and c1_above_30
            // Rule 3: big bullish Marubozu + small bearish candle (<= 30% of C1, low volume) holding its upper half
            bool r3 = is_big_maru[1] and is_bull[1] and is_bear and (c_range <= bo_c2_max_size * c_range[1] + eps) and (low > mid_c1) and c2_low_sell and c1_above_30
            ok := r1 or r2 or r3

        else if dir == -1
            bool r1 = is_marubozu[1] and is_marubozu and is_bear[1] and is_bear and (close[1] < level) and (close < level) and (close < low[1]) and (c_range >= bo_c2_min_r1 * c_range[1] - eps)
            // More than 30% of C1's WHOLE candle (body + shadows) below the CTS
            float c1_part_below = math.max(0.0, math.min(high[1], level) - low[1])
            bool c1_below_30 = c1_part_below > bo_c1_above_pct * c_range[1] + eps

            float mid_c1 = (high[1] + low[1]) / 2.0
            bool r2 = is_big_maru[1] and is_bear[1] and is_bear and (high < mid_c1) and c2_low_buy and c1_below_30
            bool r3 = is_big_maru[1] and is_bear[1] and is_bull and (c_range <= bo_c2_max_size * c_range[1] + eps) and (high < mid_c1) and c2_low_buy and c1_below_30
            ok := r1 or r2 or r3
    // ...or one of the special three-candle patterns (A, B, C)
    bool triOk = f_tri_at(level, dir)
    ok or triOk

// =====================================================================
// 100% REAL PULLBACK CONFIRMATION (3 MAIN CONDITIONS)
// =====================================================================
float mid_1 = (nz(high[1], high) + nz(low[1], low)) / 2.0

// The three Real Pullback conditions (identical logic in up- and downtrends).
// All sizes compare the TOTAL candle length (high - low), never the body.
// Rule 1: two consecutive counter-trend Marubozus; C2 closes beyond C1's
//         LOW (bearish) / HIGH (bullish) and C2's length is at least 70% of C1's.
// Rule 2: BIG counter-trend Marubozu + a candle (any size) in the SAME
//         direction whose far end stays short of C1's 50% level
//         (bearish: C2 high < C1 mid; bullish: C2 low > C1 mid), low volume on C2.
// Rule 3: BIG counter-trend Marubozu + a small OPPOSITE candle
//         whose length is 30% or less of C1's, with its far end short of
//         C1's 50% level, low volume on C2.
// BIG = larger than at least 3 of the 5 preceding Marubozu candles.

// BEARISH PULLBACK (Counter-trend down in an Uptrend)
bool pb_r1_dn = is_bear[1] and is_marubozu[1] and is_bear and is_marubozu and (c_range >= pb_c2_min_size * nz(c_range[1], c_range) - eps) and (close < nz(low[1], low))
bool pb_r2_dn = is_bear[1] and is_big_maru[1] and is_bear and (high < mid_1) and c2_low_buy
bool pb_r3_dn = is_bear[1] and is_big_maru[1] and is_bull and (c_range <= pb_c2_max_r3 * nz(c_range[1], c_range) + eps) and (high < mid_1) and c2_low_buy

// ...or a special three-candle pattern (A, B, C) against the trend.
bool pb_pattern_down = pb_r1_dn or pb_r2_dn or pb_r3_dn or tri_dn

// BULLISH PULLBACK (Counter-trend up in a Downtrend)
bool pb_r1_up = is_bull[1] and is_marubozu[1] and is_bull and is_marubozu and (c_range >= pb_c2_min_size * nz(c_range[1], c_range) - eps) and (close > nz(high[1], high))
bool pb_r2_up = is_bull[1] and is_big_maru[1] and is_bull and (low > mid_1) and c2_low_sell
bool pb_r3_up = is_bull[1] and is_big_maru[1] and is_bear and (c_range <= pb_c2_max_r3 * nz(c_range[1], c_range) + eps) and (low > mid_1) and c2_low_sell

bool pb_pattern_up = pb_r1_up or pb_r2_up or pb_r3_up or tri_up

// Real Breakout SHAPES without the level test (used by the pattern markers).
bool mk_r1_up = is_marubozu[1] and is_marubozu and is_bull[1] and is_bull and close > high[1] and c_range >= bo_c2_min_r1 * c_range[1] - eps
bool mk_r2_up = is_big_maru[1] and is_bull[1] and is_bull and low > mid_1 and c2_low_sell
bool mk_r3_up = is_big_maru[1] and is_bull[1] and is_bear and (c_range <= bo_c2_max_size * c_range[1] + eps) and low > mid_1 and c2_low_sell
bool mk_r1_dn = is_marubozu[1] and is_marubozu and is_bear[1] and is_bear and close < low[1] and c_range >= bo_c2_min_r1 * c_range[1] - eps
bool mk_r2_dn = is_big_maru[1] and is_bear[1] and is_bear and high < mid_1 and c2_low_buy
bool mk_r3_dn = is_big_maru[1] and is_bear[1] and is_bull and (c_range <= bo_c2_max_size * c_range[1] + eps) and high < mid_1 and c2_low_buy
bool shapeBoUp = mk_r1_up or mk_r2_up or mk_r3_up or tri_up
bool shapeBoDn = mk_r1_dn or mk_r2_dn or mk_r3_dn or tri_dn

var int msTrendState = 0

// ---- Market-structure state (read by the Range Box and the structure engine) ----
// Up side: impulse up (H*) / bearish pullback (L*)
var float uExt    = na
var int   uExtBar = na
var float uFake    = na   // highest fake high beyond the locked CTS (FBO, no Real Breakout)
var int   uFakeBar = na
var float dFake    = na   // lowest fake low beyond the locked CTS (FBO, no Real Breakout)
var int   dFakeBar = na
var bool  uLock   = false
var float uPbx    = na
var int   uPbxBar = na
// Down side: impulse down (L*) / bullish pullback (H*)
var float dExt    = na
var int   dExtBar = na
var bool  dLock   = false
var float dPbx    = na
var int   dPbxBar = na

// Level whose Real Breakout reverses the trend: the last BOS, or A right after
// a reversal (A is not labelled).
var float keyLevel = na
var int   keyBar   = na
var line  keyLine  = na     // drawn BOS line, extended until it is broken or replaced


// =====================================================================
// PRICE ACTION RANGE BOX ENGINE
// ---------------------------------------------------------------------
// State machine:
//   IDLE   --(one of the 4 range types, no Real Pullback on the start candles)--> ACTIVE
//   ACTIVE --(trend-side Real Breakout, Rules 1-3)------------------> CLOSED (Scenario 2)
//   ACTIVE --(counter-trend Real Pullback, Rules 1-3, anywhere)----> CLOSED (Scenario 1)
//   ACTIVE --(counter-side Real Breakout, Rules 1-3)----------------> CLOSED (Scenario 1)
//   ACTIVE --(anything else: inside candles, wicks, body whipsaws)--> ACTIVE (expanded by Fakeouts)
//
// BOUNDARY MECHANICS (V15.4) - the box only ends on Rules 1-3:
//   * The box closes ONLY on one of the three Real Breakout or Real Pullback
//     conditions, however many candles it lasts.
//   * Every candle that leaves the box (by wick OR body) without completing a
//     Real Breakout / Real Pullback belongs to the range: the box expands up,
//     down and to the right to contain it.
//   * While candles CLOSE beyond a boundary, that boundary is held so every
//     consecutive two-candle pattern is tested against it (V15.4 absorbed the
//     second candle of a failed pair, which skipped a Real Breakout starting on
//     that candle and let a steep trend move be swallowed). When price closes
//     back inside with no Real Breakout / Pullback, the whole excursion is
//     absorbed and the box expands.
//
// While ACTIVE every candle is internal liquidity: no pullback is
// confirmed, no CTS/BOS is issued, no signal dot is plotted. Only the
// candle that closes the box may carry a signal.
// =====================================================================
var bool  rangeActive   = false
var int   rangeDir      = 0      // +1 box in an up leg (C1 bearish), -1 box in a down leg (C1 bullish)
var int   rangeSeedBar  = na     // bar on which the box was confirmed (end of its start candles)
var int   rangeType     = 0      // 1 = trapped (types 1/2), 3 = invalid pullback, 4 = fake breakout (FBO)
var float boxTop        = na     // ceiling: C1/C2 extremes + absorbed wick sweeps and Fakeouts
var float boxBot        = na     // floor:   C1/C2 extremes + absorbed wick sweeps and Fakeouts
var int   outUpBars     = 0      // consecutive closes above the ceiling (excursion in progress)
var int   outDnBars     = 0      // consecutive closes below the floor (excursion in progress)
var float exHi          = na     // highest high of the current upside excursion
var float exLo          = na     // lowest low of the current downside excursion
var int   rangeLeftBar  = na     // bar of Candle 1 (the anchor candle, always retained)
var box   rangeBox      = na
// Drawn type 1-3 range boxes and their anchor labels (for the correction-zone filter)
var box[]   rbBoxes  = array.new_box()
var label[] rbLabels = array.new_label()
var int[]   rbLeft   = array.new_int()
var int   lastCloseBar  = na     // bar on which the last box closed
var int   seedBias      = 0      // direction of the last real move (+1 up / -1 down), 0 = use msTrendState

// Closure tests against the box as it stood at the START of this bar (frozen,
// after all previous expansions). Evaluated on every bar (not inside a
// conditional) so the history-referencing functions stay consistent.
float ceilLvl  = rangeActive ? boxTop : na
float floorLvl = rangeActive ? boxBot : na
bool boUpCeil  = f_real_breakout(ceilLvl, 1)        // Real Breakout above the ceiling
bool boDnFloor = f_real_breakout(floorLvl, -1)      // Real Breakout below the floor
// Range-exit Real Pullback (sections 3-3 item 2 and 4-5): one of the 6 patterns
// against the box's leg with line = the current floor / ceiling = a Real
// Breakout of that boundary against the leg (types 1-3).
bool pbDnExit  = rangeActive and rangeType != 4 and rangeDir == 1  and boDnFloor
bool pbUpExit  = rangeActive and rangeType != 4 and rangeDir == -1 and boUpCeil

// ---------------- Per-bar Range events ----------------
bool rangeWasActive = rangeActive
bool rngBreakUp     = false   // Scenario 2 closure, up-leg box broken above its ceiling
bool rngBreakDn     = false   // Scenario 2 closure, down-leg box broken below its floor
bool rngPbClose     = false   // Scenario 1 closure, Real Pullback against the leg
bool rngClosedNow   = false
int  rngClosedDir   = 0       // direction of the box that closed on this bar
bool rngBoxNew      = false   // an FBO box was created on this bar
bool rangeSeed      = false

// Start-candle exception (section 3-3 item 3): the FIRST candle of a Real
// Pullback pattern completed now is one of the candles the box was formed with
// (up to the bar it was confirmed on) -> it was the pullback, not a range.
bool pb2Dn   = pb_r1_dn or pb_r2_dn or pb_r3_dn
bool pb2Up   = pb_r1_up or pb_r2_up or pb_r3_up
bool startDn = (pb2Dn and bar_index - 1 <= rangeSeedBar) or (tri_dn and bar_index - 2 <= rangeSeedBar)
bool startUp = (pb2Up and bar_index - 1 <= rangeSeedBar) or (tri_up and bar_index - 2 <= rangeSeedBar)
bool rngCancelNow = rangeActive and rangeType != 4 and (rangeDir == 1 ? startDn : startUp)
bool rngCancelDn  = rngCancelNow and rangeDir == 1    // up-leg box: bearish Real Pullback
bool rngCancelUp  = rngCancelNow and rangeDir == -1   // down-leg box: bullish Real Pullback

// Type 3 reference ceiling / floor, read BEFORE this bar's structure update:
// wave not locked -> the highest High of the current wave; locked -> High of C1.
float waveHigh = msTrendState >= 0 and not na(uExt) and not uLock ? math.max(uExt, nz(high[1], high)) : nz(high[1], high)
float waveLow  = msTrendState <= 0 and not na(dExt) and not dLock ? math.min(dExt, nz(low[1], low))   : nz(low[1], low)

// =====================================================================
// ORDER OF EVENTS ON EVERY CLOSED CANDLE (section 4-2)
//   1. CTS and trend change (market structure)        - then the Hide filter
//   2. start-candle exception: an active box 1-3 is removed
//   3. Fake Breakout (FBO)
//   4. an active box: expansion and end (Real Breakout / Real Pullback)
//   5. no active box: start of a range, type 3 first, else type 1 / 2
// =====================================================================

// =====================================================================
// MARKET STRUCTURE (CTS / BOS) - implements Market_Structure_Prompt_FA.md
// ---------------------------------------------------------------------
// Up-trend cycle (down-trend is the exact mirror):
//   PHASE A (impulse)  : track H* = highest high of the up impulse. A counter
//                        move WITHOUT one of the 3 Real Pullback conditions is a
//                        range/noise: H* just keeps extending (no CTS, no BOS).
//   -> one of the 3 Real Pullback conditions (bearish) occurs
//   PHASE B (pullback) : H* is LOCKED (pending CTS). Track L* = lowest low of the
//                        WHOLE pullback wave since H*, however long it lasts and
//                        whatever ranges form inside it. Fake moves above H* do
//                        not move it; fake moves below the last BOS do not
//                        reverse the trend.
//   -> Real Breakout (3 conditions) of H*
//        CTS (Recent High) = H*  and  BOS (Key Level) = L*, identified TOGETHER
//        at this bar; a new impulse starts (back to PHASE A).
//   Real Breakout of the last BOS downward (in PHASE A or B) -> trend DOWN.
//        A = H* (highest high of the last up wave) is the unlabelled reversal
//        level until the first bearish BOS is identified.
// Chart start: the trend is undetermined; both directions run the cycle and
// the first complete cycle (impulse -> Real Pullback -> Real Breakout) sets it.
// Every timeframe runs these rules on its own candles.
// =====================================================================
bool is_confirmed = msStrictClose ? barstate.isconfirmed : true

// Real Pullback for STRUCTURE (section 3-3).
//  - no active range: any of the 6 Real Pullback patterns;
//  - active range: only one of the 6 patterns breaking the box against its leg
//    (line = current floor / ceiling, same line tests as a Real Breakout);
//  - exception: the start candles of the box complete a Real Pullback.
bool pbDnStruct = (pb_pattern_down and (not rangeWasActive or rngCancelDn)) or pbDnExit
bool pbUpStruct = (pb_pattern_up   and (not rangeWasActive or rngCancelUp)) or pbUpExit

// Real Breakout tests, evaluated on every bar (levels are the previous bar's state)
bool revDnBreak = f_real_breakout(msTrendState == 1 ? keyLevel : na, -1)
bool revUpBreak = f_real_breakout(msTrendState == -1 ? keyLevel : na, 1)
// While a Fake Breakout (type 4) box older than 2 bars is active (start candle = #1,
// so from candle #3 on), the pending CTS is broken only by a Real Breakout of the
// box's far side (the fake extreme); the CTS itself is still drawn on its line.
bool  fboBoxOld = rangeActive and rangeType == 4 and bar_index - rangeLeftBar >= 2
float ctsUpLvl  = uLock ? (fboBoxOld and rangeDir == 1  ? math.max(uExt, boxTop) : uExt) : na
float ctsDnLvl  = dLock ? (fboBoxOld and rangeDir == -1 ? math.min(dExt, boxBot) : dExt) : na
bool ctsUpBreak = f_real_breakout(ctsUpLvl, 1)
bool ctsDnBreak = f_real_breakout(ctsDnLvl, -1)

// Pending CTS level (for the Fake Breakout marker), captured before the update
int   fbTrendPre = msTrendState
float fbLevelPre = msTrendState == 1 and uLock ? uExt : msTrendState == -1 and dLock ? dExt : na

msSignal = 0            // 1 = CTS up, -1 = CTS down, 2 = trend -> up, -2 = trend -> down
bool  pbLockUp = false  // bearish Real Pullback locked the up-side CTS candidate on this bar
bool  pbLockDn = false  // bullish Real Pullback locked the down-side CTS candidate on this bar
float ctsPrice = na
int   ctsBar   = na
float bosPrice = na
int   bosBar   = na
float revPrice = na     // A: reversal level after a trend change
int   revBar   = na

int trendPre = msTrendState
if is_confirmed
    if trendPre == 1 and revDnBreak
        // ---- Trend reversal: last BOS broken downward by a Real Breakout ----
        msSignal := -2
        msTrendState := -1
        revPrice := uExt
        revBar   := uExtBar
        if na(uPbx) or low < uPbx
            dExt    := low
            dExtBar := bar_index
        else
            dExt    := uPbx
            dExtBar := uPbxBar
        dLock    := false
        dPbx     := na
        dPbxBar  := na
        dFake    := na
        dFakeBar := na
        keyLevel := uExt
        keyBar   := uExtBar
    else if trendPre == -1 and revUpBreak
        // ---- Trend reversal: last BOS broken upward by a Real Breakout ----
        msSignal := 2
        msTrendState := 1
        revPrice := dExt
        revBar   := dExtBar
        if na(dPbx) or high > dPbx
            uExt    := high
            uExtBar := bar_index
        else
            uExt    := dPbx
            uExtBar := dPbxBar
        uLock    := false
        uPbx     := na
        uPbxBar  := na
        uFake    := na
        uFakeBar := na
        keyLevel := dExt
        keyBar   := dExtBar
    else
        // ---- Up side (active in an uptrend or while the trend is undetermined) ----
        if trendPre >= 0
            if na(uExt)
                uExt    := high
                uExtBar := bar_index
            else if uLock
                if ctsUpBreak
                    msSignal := 1
                    if msTrendState == 0
                        msTrendState := 1
                    ctsPrice := uExt
                    ctsBar   := uExtBar
                    bosPrice := uPbx
                    bosBar   := uPbxBar
                    keyLevel := uPbx
                    keyBar   := uPbxBar
                    uExt     := high
                    uExtBar  := bar_index
                    uLock    := false
                    uPbx     := na
                    uPbxBar  := na
                    uFake    := na
                    uFakeBar := na
                else if not na(uFake) and not na(uPbx) and low < uPbx
                    // State 2: no Real Breakout, only fakes beyond the CTS, and the
                    // correction now breaks the previous pullback -> that pullback is
                    // dropped; the CTS moves to the highest fake high and this new
                    // correction is the pullback (CTS / BOS wait for its Real Breakout).
                    uExt     := uFake
                    uExtBar  := uFakeBar
                    uPbx     := low
                    uPbxBar  := bar_index
                    uFake    := na
                    uFakeBar := na
                else
                    if na(uPbx) or low < uPbx
                        uPbx    := low
                        uPbxBar := bar_index
                    if high > uExt and (na(uFake) or high > uFake)
                        uFake    := high
                        uFakeBar := bar_index
            else
                if high > uExt
                    uExt    := high
                    uExtBar := bar_index
                    uPbx    := na
                    uPbxBar := na
                else if na(uPbx) or low < uPbx
                    uPbx    := low
                    uPbxBar := bar_index
                if pbDnStruct
                    uLock    := true
                    pbLockUp := true
                    uFake    := na
                    uFakeBar := na
                    if na(uPbx)
                        uPbx    := low
                        uPbxBar := bar_index
        // ---- Down side (active in a downtrend or while the trend is undetermined) ----
        if trendPre <= 0
            if na(dExt)
                dExt    := low
                dExtBar := bar_index
            else if dLock
                if ctsDnBreak
                    msSignal := -1
                    if msTrendState == 0
                        msTrendState := -1
                    ctsPrice := dExt
                    ctsBar   := dExtBar
                    bosPrice := dPbx
                    bosBar   := dPbxBar
                    keyLevel := dPbx
                    keyBar   := dPbxBar
                    dExt     := low
                    dExtBar  := bar_index
                    dLock    := false
                    dPbx     := na
                    dPbxBar  := na
                    dFake    := na
                    dFakeBar := na
                else if not na(dFake) and not na(dPbx) and high > dPbx
                    // State 2 (mirror): fakes below the CTS, then the correction breaks
                    // the previous pullback -> CTS moves to the lowest fake low and this
                    // new correction is the pullback.
                    dExt     := dFake
                    dExtBar  := dFakeBar
                    dPbx     := high
                    dPbxBar  := bar_index
                    dFake    := na
                    dFakeBar := na
                else
                    if na(dPbx) or high > dPbx
                        dPbx    := high
                        dPbxBar := bar_index
                    if low < dExt and (na(dFake) or low < dFake)
                        dFake    := low
                        dFakeBar := bar_index
            else
                if low < dExt
                    dExt    := low
                    dExtBar := bar_index
                    dPbx    := na
                    dPbxBar := na
                else if na(dPbx) or high > dPbx
                    dPbx    := high
                    dPbxBar := bar_index
                if pbUpStruct
                    dLock    := true
                    pbLockDn := true
                    dFake    := na
                    dFakeBar := na
                    if na(dPbx)
                        dPbx    := high
                        dPbxBar := bar_index

bull_break_confirmed = (msSignal == 1) or (msSignal == 2)
bear_break_confirmed = (msSignal == -1) or (msSignal == -2)

// ---- STEP 1b: correction-zone filter (section 4-6) ----
// On a confirmed CTS the correction (from the wave top at ctsBar to this
// breakout) is known: the type 1-3 ranges that started inside it are deleted
// with their labels, including one still open. FBO boxes are never deleted.
// After that no box is active and a new range starts after this candle.
if rngHideInCorrection and math.abs(msSignal) == 1 and not na(ctsBar) and array.size(rbBoxes) > 0
    for i = array.size(rbBoxes) - 1 to 0
        int rbL = array.get(rbLeft, i)
        if rbL >= ctsBar
            box   bx = array.get(rbBoxes, i)
            label lb = array.get(rbLabels, i)
            if not na(bx)
                box.delete(bx)
            if not na(lb)
                label.delete(lb)
            if rangeActive and rbL == rangeLeftBar and rangeType != 4
                rangeBox     := na
                rangeActive  := false
                rangeDir     := 0
                rangeType    := 0
                outUpBars    := 0
                outDnBars    := 0
                exHi         := na
                exLo         := na
                lastCloseBar := bar_index
            array.remove(rbBoxes, i)
            array.remove(rbLabels, i)
            array.remove(rbLeft, i)

// ---- STEP 2: start-candle exception (section 3-3 item 3) ----
if rngCancelNow and rangeActive and rangeType != 4
    if not na(rangeBox)
        box.delete(rangeBox)
    rangeBox     := na
    rangeActive  := false
    rangeDir     := 0
    rangeType    := 0
    outUpBars    := 0
    outDnBars    := 0
    exHi         := na
    exLo         := na
    lastCloseBar := bar_index

// ---- STEP 3: Fake Breakout ----
// =====================================================================
// FAKE BREAKOUT (FBO)
// START ("FBO in progress"): the first candle that trades beyond the pending
// CTS level (wick or close) while none of the 3 main Real Breakout conditions
// and none of the 3 three-candle patterns completes. Later candles beyond the
// line (inside bars or anything else) keep it in progress.
// CONFIRMED: price closes back on the other side of the CTS line (can be the
// start candle itself when it only crossed with a wick).
// A Real Breakout completing while the FBO is still in progress means the move
// was not fake: the grey label is deleted. A confirmed FBO label stays even if
// a Real Breakout follows later. Shown inside ranges as well.
// Display: an "FBO" label on the CTS line (grey = in progress, red = confirmed).
// =====================================================================
var bool  fbActive = false
var float fbLevel  = na
var int   fbDir    = 0
var label fbLabel  = na
// A structural signal (CTS / trend change) ends a Fake Breakout (type 4) box:
// in candle #1 or #2 of the box, or while the FBO is still unconfirmed (grey),
// the move was not fake -> the box is removed; otherwise it closes.
if rangeActive and rangeType == 4 and msSignal != 0
    if bar_index - rangeLeftBar <= 1 or fbActive
        if not na(rangeBox)
            box.delete(rangeBox)
    else if not na(rangeBox)
        box.set_right(rangeBox, bar_index)
        box.set_bgcolor(rangeBox, msSignal > 0 ? rngBullColor : rngBearColor)
        box.set_border_color(rangeBox, color.new(msSignal > 0 ? rngBullColor : rngBearColor, 15))
        box.set_border_style(rangeBox, line.style_solid)
        box.set_text(rangeBox, "Range Closed")
        box.set_text_size(rangeBox, size.tiny)
        box.set_text_color(rangeBox, color.new(msSignal > 0 ? rngBullColor : rngBearColor, 0))
        box.set_text_halign(rangeBox, text.align_left)
    rangeBox     := na
    rangeActive  := false
    rangeDir     := 0
    rangeType    := 0
    outUpBars    := 0
    outDnBars    := 0
    exHi         := na
    exLo         := na
    lastCloseBar := bar_index
if fbActive and (msSignal != 0 or na(fbLevelPre) or fbLevelPre != fbLevel)
    // Real Breakout / trend change / new CTS while in progress: not a fake
    if not na(fbLabel)
        label.delete(fbLabel)
    fbLabel  := na
    fbActive := false
// One FBO per CTS line: later crossings of the same line belong to the same FBO.
var float fbDoneLevel = na
bool fbAttempt = not na(fbLevelPre) and msSignal == 0 and (na(fbDoneLevel) or fbLevelPre != fbDoneLevel) and (fbTrendPre == 1 ? high > fbLevelPre : low < fbLevelPre)
bool fakeBreakoutStart = fbAttempt and not fbActive and is_confirmed
if fakeBreakoutStart
    fbActive := true
    fbLevel  := fbLevelPre
    fbDir    := fbTrendPre
    // An FBO always forms its own box: a type 1-3 range still active is closed here.
    if rangeActive and rangeType != 4
        if not na(rangeBox)
            box.set_right(rangeBox, bar_index)
            box.set_bgcolor(rangeBox, rngPbColor)
            box.set_border_color(rangeBox, color.new(rngPbColor, 15))
            box.set_border_style(rangeBox, line.style_solid)
            box.set_text(rangeBox, "Range Closed")
            box.set_text_size(rangeBox, size.tiny)
            box.set_text_color(rangeBox, color.new(rngPbColor, 0))
            box.set_text_halign(rangeBox, text.align_left)
        rangeBox     := na
        rangeActive  := false
        rangeDir     := 0
        rangeType    := 0
        outUpBars    := 0
        outDnBars    := 0
        exHi         := na
        exLo         := na
        lastCloseBar := bar_index
    // Range type 4: box between the CTS line (fixed side) and the fake candle's
    // extreme; it grows only away from the CTS until a Real Breakout.
    if not rangeActive
        rangeActive  := true
        rangeType    := 4
        rangeDir     := fbDir
        boxTop       := fbDir == 1 ? high : fbLevel
        boxBot       := fbDir == 1 ? fbLevel : low
        rangeLeftBar := bar_index
        outUpBars    := 0
        outDnBars    := 0
        exHi         := na
        exLo         := na
        rngBoxNew    := true
        if rngShow
            rangeBox := box.new(left=bar_index, top=boxTop, right=bar_index, bottom=boxBot, bgcolor=rngActiveColor, border_color=color.new(color.gray, 25), border_width=1, border_style=line.style_dotted)
    if msShowFakeBO
        fbLabel := label.new(bar_index, fbLevel, "FBO", style=fbDir == 1 ? label.style_label_down : label.style_label_up, color=color.new(color.gray, 80), textcolor=color.gray, size=size.tiny, tooltip="Fake Breakout in progress: price beyond the CTS without any Real Breakout condition or 3-candle pattern")
bool fakeBreakoutConfirmed = false
if fbActive and is_confirmed and msSignal == 0 and (fbDir == 1 ? close < fbLevel : close > fbLevel)
    fakeBreakoutConfirmed := true
    fbActive    := false
    fbDoneLevel := fbLevel
    if not na(fbLabel)
        label.set_color(fbLabel, color.new(color.red, 80))
        label.set_textcolor(fbLabel, color.red)
        label.set_tooltip(fbLabel, "Fake Breakout confirmed: price closed back on the other side of the CTS")
    fbLabel := na

// ---- STEP 4: an active box (one that existed at the start of the bar) ----
if rangeActive and rangeWasActive and not rngBoxNew
    // Scenario 2 = Real Breakout of the trend-side boundary:
    //   up-leg box   -> above the ceiling
    //   down-leg box -> below the floor
    bool realBO = rangeDir == 1 ? boUpCeil : boDnFloor
    // Scenario 1 (section 4-5): one of the 6 patterns AGAINST the leg with line =
    // the current floor (up leg) / ceiling (down leg) -> orange close, the
    // pullback counts (pbDnExit / pbUpExit feed the structure).
    // Type 4 (FBO) box (section 5-2): ends by a Real Breakout of its far side or
    // of its near side (the CTS line) in the opposite direction.
    bool realPB = rangeType == 4 ? (rangeDir == 1 ? boDnFloor : boUpCeil) : rangeDir == 1 ? pbDnExit : pbUpExit

    if realBO
        // ---- SCENARIO 2: Real Breakout closes the Range Box ----
        rngBreakUp    := rangeDir == 1
        rngBreakDn    := rangeDir == -1
        rngClosedNow  := true
        seedBias      := rangeDir
    else if realPB
        // ---- SCENARIO 1: Real Pullback closes the Range Box ----
        rngPbClose   := true
        rngClosedNow := true
        seedBias     := -rangeDir
    else
        // ---- NO CLOSURE: the box persists (see BOUNDARY MECHANICS) ----
        // Type 4 (FBO) box: the CTS side stays ON the CTS line; it only grows away from it.
        bool fixCeil  = rangeType == 4 and rangeDir == -1
        bool fixFloor = rangeType == 4 and rangeDir == 1
        // Ceiling side
        if close > boxTop
            // Close above the ceiling: keep the ceiling so EVERY consecutive pattern
            // is tested against it (none is skipped), record the excursion.
            outUpBars += 1
            exHi      := math.max(nz(exHi, high), high)
        else
            // Closed back inside without a Real Breakout / Pullback: the whole
            // excursion (and any wick) belongs to the range -> expand.
            if not fixCeil
                boxTop := math.max(boxTop, nz(exHi, high), high)
            outUpBars := 0
            exHi      := na
        // Floor side (mirror)
        if close < boxBot
            outDnBars += 1
            exLo      := math.min(nz(exLo, low), low)
        else
            if not fixFloor
                boxBot := math.min(boxBot, nz(exLo, low), low)
            outDnBars := 0
            exLo      := na

        // Box grows rightward and vertically.
        if not na(rangeBox)
            box.set_right(rangeBox, bar_index)
            box.set_top(rangeBox, boxTop)
            box.set_bottom(rangeBox, boxBot)

    if rngClosedNow

        outDnBars    := 0

        exLo         := na

// ---- STEP 5: start of a range (only when no box is active) ----

// Type 3 - invalid pullback (d = +1; mirror for -1): C1 counter to d, C2 does not
// close beyond the reference ceiling, no Real Pullback against d completed on C1
// or C2, C1 after the bar the previous range closed on. Box from C1, bounds = C1/C2.
bool c1AfterExit  = na(lastCloseBar) or (bar_index - 1) > lastCloseBar
bool noPullbackUp = not pb_pattern_down and not pb_pattern_down[1]   // no bearish Real Pullback on C1 / C2
bool noPullbackDn = not pb_pattern_up   and not pb_pattern_up[1]     // no bullish Real Pullback on C1 / C2
bool seed3Up = legDir == 1  and c1AfterExit and bar_index >= 1 and close[1] < open[1] and close <= waveHigh and noPullbackUp
bool seed3Dn = legDir == -1 and c1AfterExit and bar_index >= 1 and close[1] > open[1] and close >= waveLow  and noPullbackDn

// Type 1 / 2 - trapping candle: a window of 3 to 5 consecutive candles ending now
// (the longest that works). The anchor is a Big Marubozu (type 1) or a Big Pin Bar
// (type 2) inside the window, not its last candle; the closes of ALL other window
// candles stay inside the anchor's Low-High (inclusive; wicks may poke out). No
// Real Pullback pattern (either direction) on the last two candles; the first
// window candle comes after the bar the previous range closed on.
int  w12Len    = 0
int  w12Anchor = 0
bool w12Pin    = false
for W = 5 to 3
    if w12Len == 0 and bar_index >= W - 1 and (na(lastCloseBar) or bar_index - (W - 1) > lastCloseBar)
        for a = W - 1 to 1
            if w12Len == 0 and (is_big_maru[a] or is_big_pin[a])
                bool held = true
                for k = 0 to W - 1
                    if k != a and (close[k] > high[a] or close[k] < low[a])
                        held := false
                if held
                    w12Len    := W
                    w12Anchor := a
                    w12Pin    := not is_big_maru[a]
bool seed12 = w12Len > 0 and noPullbackUp and noPullbackDn
float w12Top = na
float w12Bot = na
if seed12
    for k = 0 to w12Len - 1
        w12Top := na(w12Top) ? high[k] : math.max(w12Top, high[k])
        w12Bot := na(w12Bot) ? low[k]  : math.min(w12Bot, low[k])

if not rangeActive and (seed3Up or seed3Dn)
    rangeSeed    := true
    rangeActive  := true
    rangeType    := 3
    rangeDir     := seed3Up ? 1 : -1
    boxTop       := math.max(high[1], high)
    boxBot       := math.min(low[1], low)
    rangeLeftBar := bar_index - 1
else if not rangeActive and seed12
    rangeSeed    := true
    rangeActive  := true
    rangeType    := w12Pin ? 2 : 1
    rangeDir     := legDir != 0 ? legDir : (close[w12Anchor] >= open[w12Anchor] ? 1 : -1)
    boxTop       := w12Top
    boxBot       := w12Bot
    rangeLeftBar := bar_index - (w12Len - 1)
if rangeSeed
    rangeSeedBar := bar_index
    outUpBars := 0
    outDnBars := 0
    exHi      := na
    exLo      := na
    if rngShow
        rangeBox := box.new(left=f_x(rangeLeftBar), top=boxTop, right=bar_index, bottom=boxBot, bgcolor=rngActiveColor, border_color=color.new(color.gray, 25), border_width=1, border_style=line.style_dotted)
    label rbLab = na
    if rngMarkC1
        // Mark the range anchor (start candle) on the chart.
        rbLab := label.new(f_x(rangeLeftBar), rangeDir == 1 ? high[bar_index - rangeLeftBar] : low[bar_index - rangeLeftBar], str.tostring(rangeType), style=label.style_none, textcolor=color.gray, size=size.tiny, yloc=rangeDir == 1 ? yloc.abovebar : yloc.belowbar)
    array.push(rbBoxes, rangeBox)
    array.push(rbLabels, rbLab)
    array.push(rbLeft, rangeLeftBar)
    if array.size(rbBoxes) > 300
        array.shift(rbBoxes)
        array.shift(rbLabels)
        array.shift(rbLeft)

bool rngRealDown = rngBreakDn or (rngPbClose and rngClosedDir == 1)
