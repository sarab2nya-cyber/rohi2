var float[] maru_sizes = array.new_float(0)

int bigger_count = 0
if array.size(maru_sizes) > 0

        if c_range > array.get(maru_sizes, i)

// =====================================================================
// REAL BREAKOUT (3 Main Conditions) - used by market structure AND the Range Box
// A breakout of `level` (the BOI line) is REAL only when it matches one of the
// three main conditions below. Anything else that crosses the level is a
// Fakeout. Every rule is a closed two-candle pattern (C1 = previous candle,
// C2 = current candle) evaluated on the bar it completes.
//
// Rule 1: C1 and C2 are Marubozus in the breakout direction, both close beyond
//         the BOI, and C2 closes beyond the ENTIRE C1 candle (its high in an
//         up-breakout, its low in a down-breakout). No % beyond BOI required.
// Rule 2: C1 is a BIG Marubozu with MORE than 30% of its body beyond the BOI;
//         C2 is a small candle in the breakout direction whose low (high for
//         down-breakouts) stays beyond the 50% mark of C1.
// Rule 3: as Rule 2, but C2 is opposite to the breakout direction and SMALLER
//         than 30% of C1's total size.
// =====================================================================
f_real_breakout(level, dir) =>
    bool ok = false
    if not na(level)
        if dir == 1
            // Rule 1: two bullish Marubozus, both closing above the BOI, C2 closing above C1's high
            bool r1 = is_marubozu[1] and is_marubozu and is_bull[1] and is_bull and (close[1] > level) and (close > level) and (close > high[1])
            // More than 30% of C1's WHOLE candle (body + shadows) above the CTS
            float c1_part_above = math.max(0.0, high[1] - math.max(low[1], level))
            bool c1_above_30 = c1_part_above > bo_c1_above_pct * c_range[1] + eps

            float mid_c1 = (high[1] + low[1]) / 2.0
            // Rule 2: big bullish Marubozu + bullish candle whose LOW stays above C1's 50%
            bool r2 = is_big_maru[1] and is_bull[1] and is_bull and (low > mid_c1) and c1_above_30
            // Rule 3: big bullish Marubozu + small bearish candle (< 30% of C1) holding its upper half
            bool r3 = is_big_maru[1] and is_bull[1] and is_bear and (c_range < bo_c2_max_size * c_range[1] - eps) and (low > mid_c1) and c1_above_30
            ok := r1 or r2 or r3

        else if dir == -1
            bool r1 = is_marubozu[1] and is_marubozu and is_bear[1] and is_bear and (close[1] < level) and (close < level) and (close < low[1])
            // More than 30% of C1's WHOLE candle (body + shadows) below the CTS
            float c1_part_below = math.max(0.0, math.min(high[1], level) - low[1])
            bool c1_below_30 = c1_part_below > bo_c1_above_pct * c_range[1] + eps

            float mid_c1 = (high[1] + low[1]) / 2.0
            bool r2 = is_big_maru[1] and is_bear[1] and is_bear and (high < mid_c1) and c1_below_30
            bool r3 = is_big_maru[1] and is_bear[1] and is_bull and (c_range < bo_c2_max_size * c_range[1] - eps) and (high < mid_c1) and c1_below_30
            ok := r1 or r2 or r3
    ok

// =====================================================================
// 100% REAL PULLBACK CONFIRMATION (3 MAIN CONDITIONS)
// =====================================================================
float mid_1 = (nz(high[1], high) + nz(low[1], low)) / 2.0

// The three Real Pullback conditions (identical logic in up- and downtrends).
// All sizes compare the TOTAL candle length (high - low), never the body.
// Only these three conditions define a Real Pullback (no volume filter).
// Rule 1: two consecutive counter-trend Marubozus; C2
//         closes beyond C1's close and C2's length is at least 70% of C1's.
// Rule 2: BIG counter-trend Marubozu + a small candle in the
//         SAME direction whose far end stays short of C1's 50% level
//         (bearish: C2 high < C1 mid; bullish: C2 low > C1 mid).
// Rule 3: BIG counter-trend Marubozu + a small OPPOSITE candle
//         whose length is 30% or less of C1's, with its far end short of
//         C1's 50% level.
// BIG = larger than at least 3 of the 5 preceding Marubozu candles.

// BEARISH PULLBACK (Counter-trend down in an Uptrend)
bool pb_r1_dn = is_bear[1] and is_marubozu[1] and is_bear and is_marubozu and (c_range >= pb_c2_min_size * nz(c_range[1], c_range) - eps) and (close < nz(close[1], close))
bool pb_r2_dn = is_bear[1] and is_big_maru[1] and is_bear and (high < mid_1)
bool pb_r3_dn = is_bear[1] and is_big_maru[1] and is_bull and (c_range <= pb_c2_max_r3 * nz(c_range[1], c_range) + eps) and (high < mid_1)

bool pb_pattern_down = pb_r1_dn or pb_r2_dn or pb_r3_dn

// BULLISH PULLBACK (Counter-trend up in a Downtrend)
bool pb_r1_up = is_bull[1] and is_marubozu[1] and is_bull and is_marubozu and (c_range >= pb_c2_min_size * nz(c_range[1], c_range) - eps) and (close > nz(close[1], close))
bool pb_r2_up = is_bull[1] and is_big_maru[1] and is_bull and (low > mid_1)
bool pb_r3_up = is_bull[1] and is_big_maru[1] and is_bear and (c_range <= pb_c2_max_r3 * nz(c_range[1], c_range) + eps) and (low > mid_1)

bool pb_pattern_up = pb_r1_up or pb_r2_up or pb_r3_up

// Real Pullback that BREAKS a range boundary against the range's leg:
//   Rule 1: both candles close beyond the boundary
//   Rule 2/3: the big Marubozu (C1 of the pattern) closes beyond the boundary
f_real_pullback_at(level, dir) =>
    bool ok = false
    if not na(level)
        if dir == -1
            ok := (pb_r1_dn and close[1] < level and close < level) or ((pb_r2_dn or pb_r3_dn) and close[1] < level)
        else if dir == 1
            ok := (pb_r1_up and close[1] > level and close > level) or ((pb_r2_up or pb_r3_up) and close[1] > level)
    ok


var int msTrendState = 0

var bool  uLock   = false

// =====================================================================
// PRICE ACTION RANGE BOX ENGINE
// ---------------------------------------------------------------------
// State machine:
//   IDLE   --(C1 counter-trend + C2 indecision + no Real Pullback)--> ACTIVE
//   ACTIVE --(trend-side Real Breakout, Rules 1-3)------------------> CLOSED (Scenario 2)
//   ACTIVE --(counter-trend Real Pullback, Rules 1-3, anywhere)----> CLOSED (Scenario 1)
//   ACTIVE --(counter-side Real Breakout, Rules 1-3)----------------> CLOSED (Scenario 1)
//   ACTIVE --(anything else: inside candles, wicks, body whipsaws)--> ACTIVE (expanded by Fakeouts)
//
// BOUNDARY MECHANICS (V15.4) - the box only ends on Rules 1-3:
//   * The box closes ONLY on one of the three Real Breakout or Real Pullback
//     conditions, however many candles it lasts.
//   * Every candle that leaves the box (by wick OR body) without completing a
//     Real Breakout / Real Pullback belongs to the range: the box expands up,
//     down and to the right to contain it.
//   * While candles CLOSE beyond a boundary, that boundary is held so every
//     consecutive two-candle pattern is tested against it (V15.4 absorbed the
//     second candle of a failed pair, which skipped a Real Breakout starting on
//     that candle and let a steep trend move be swallowed). When price closes
//     back inside with no Real Breakout / Pullback, the whole excursion is
//     absorbed and the box expands.
//
// While ACTIVE every candle is internal liquidity: no pullback is
// confirmed, no CTS/BOS is issued, no signal dot is plotted. Only the
// candle that closes the box may carry a signal.
// =====================================================================
var bool  rangeActive   = false
var int   rangeDir      = 0      // +1 box in an up leg (C1 bearish), -1 box in a down leg (C1 bullish)
var float boxTop        = na     // ceiling: C1/C2 extremes + absorbed wick sweeps and Fakeouts
var float boxBot        = na     // floor:   C1/C2 extremes + absorbed wick sweeps and Fakeouts
var int   outUpBars     = 0      // consecutive closes above the ceiling (excursion in progress)
var int   outDnBars     = 0      // consecutive closes below the floor (excursion in progress)
var float exHi          = na     // highest high of the current upside excursion
var float exLo          = na     // lowest low of the current downside excursion
var int   rangeLeftBar  = na     // bar of Candle 1 (the anchor candle, always retained)
var box   rangeBox      = na
var int   lastCloseBar  = na     // bar on which the last box closed
var int   seedBias      = 0      // direction of the last real move (+1 up / -1 down), 0 = use msTrendState

// Closed Range registry (historical boundaries). A structural level whose
// price is strictly inside a closed box and whose bar lies inside its span
// can never become CTS / BOS / Key Level.
var float[] crTop   = array.new_float()
var float[] crBot   = array.new_float()
var int[]   crLeft  = array.new_int()
var int[]   crRight = array.new_int()

f_inClosedRange(float price, int barIdx) =>
    bool inside = false
    if not na(price) and not na(barIdx) and array.size(crTop) > 0
        for i = 0 to array.size(crTop) - 1
            if barIdx >= array.get(crLeft, i) and barIdx <= array.get(crRight, i) and price < array.get(crTop, i) and price > array.get(crBot, i)
                inside := true
                break
    inside

// Closure tests against the current box boundaries (which never move while
// closes are beyond them). Evaluated on every bar (not inside a
// conditional) so the history-referencing functions stay consistent.
float ceilLvl  = rangeActive ? boxTop : na
float floorLvl = rangeActive ? boxBot : na
bool boUpCeil  = f_real_breakout(ceilLvl, 1)        // Real Breakout above the ceiling
bool boDnFloor = f_real_breakout(floorLvl, -1)      // Real Breakout below the floor
bool pbDnExit  = f_real_pullback_at(floorLvl, -1)   // bearish Real Pullback breaking the floor
bool pbUpExit  = f_real_pullback_at(ceilLvl, 1)     // bullish Real Pullback breaking the ceiling

// ---------------- Candle 1 (Trigger): counter-trend candle ----------------
float c1Range   = bar_index >= 1 ? math.max(high[1] - low[1], syminfo.mintick) : syminfo.mintick
float c1Body    = bar_index >= 1 ? math.abs(close[1] - open[1]) : 0
float c1BodyPct = c1Body / c1Range
float c1Upper   = bar_index >= 1 ? high[1] - math.max(open[1], close[1]) : 0
float c1Lower   = bar_index >= 1 ? math.min(open[1], close[1]) - low[1] : 0

bool c1Bear       = bar_index >= 1 and close[1] < open[1]
bool c1Bull       = bar_index >= 1 and close[1] > open[1]
bool c1ValidBody  = c1Body > syminfo.mintick
bool c1Doji       = c1BodyPct <= rngDojiPct
bool c1Maru       = c1ValidBody and c1BodyPct > bo_maru_body
bool c1Normal     = c1ValidBody and c1BodyPct >= rngNormalMinPct and c1BodyPct <= rngNormalMaxPct
bool c1UpperPin   = c1ValidBody and c1Upper >= c1Body * rngPinWickRatio and c1Upper > c1Lower
bool c1LowerPin   = c1ValidBody and c1Lower >= c1Body * rngPinWickRatio and c1Lower > c1Upper
bool c1Pin        = c1UpperPin or c1LowerPin
bool c1FormValid  = c1Pin or c1Doji or c1Normal or c1Maru   // the 4 permitted forms

// ---------------- Candle 2 (Confirmation): indecision test ----------------
// Protrusion = part of C2's total height (body + wicks) outside C1's range.
bool  c2Inside     = bar_index >= 1 and high <= high[1] and low >= low[1]
float c2Protrude   = bar_index >= 1 ? math.max(high - high[1], 0) + math.max(low[1] - low, 0) : c_range
bool  c2Within30   = (c2Protrude / c_range) <= rngMaxOutsidePct

// C2 is valid if it satisfies ONE of the four conditions:
//  A) complete Inside Bar (body and wicks inside C1's range)
//  B) max 30% protrusion (for non-Normal / non-Marubozu candles; Normal
//     candles are governed by D, momentum Marubozus only qualify via A)
//  C) Pin Bar or Doji
//  D) Normal (50-70% body) ONLY if C1 is Pin Bar/Doji AND the 30% rule is met
bool c2CondA = c2Inside
bool c2CondB = c2Within30 and not rngNormal and not rngMaru
bool c2CondC = rngPinBar or rngDoji
bool c2CondD = rngNormal and (c1Pin or c1Doji) and c2Within30
bool c2Valid = bar_index >= 1 and (c2CondA or c2CondB or c2CondC or c2CondD)

// ---------------- Pullback check before formation ----------------
// If C1 (bar -1) or C2 (bar 0) completes a Real Pullback, it is a
// pullback, NOT a Range Box.
bool noPullbackUp = not pb_pattern_down and not pb_pattern_down[1]  // up leg: no bearish Real Pullback
bool noPullbackDn = not pb_pattern_up   and not pb_pattern_up[1]    // down leg: no bullish Real Pullback

// ---------------- Range initiation after a breakout ----------------
// "Counter-trend" is measured against the LAST REAL MOVE: after a box is
// broken downward the next box needs a bullish C1, after an upward break
// a bearish C1. Before any box has closed, the market-structure trend is
// used. C1 must be a candle AFTER the closing candle (the closing candle
// is never reused as C1), and the box only opens if C2 also qualifies.
int  legDir      = seedBias != 0 ? seedBias : msTrendState
bool c1AfterExit = na(lastCloseBar) or (bar_index - 1) > lastCloseBar
// No new range between a Real Pullback and the Real Breakout that confirms it.
bool pbPending = (msTrendState == 1 and uLock) or (msTrendState == -1 and dLock)

bool seedUp = legDir == 1  and not pbPending and c1AfterExit and c1Bear and c1FormValid and c2Valid and noPullbackUp
bool seedDn = legDir == -1 and not pbPending and c1AfterExit and c1Bull and c1FormValid and c2Valid and noPullbackDn

// ---------------- Per-bar Range events ----------------
bool rangeWasActive = rangeActive
bool rngBreakUp     = false   // Scenario 2 closure, up-leg box broken above its ceiling
bool rngBreakDn     = false   // Scenario 2 closure, down-leg box broken below its floor
bool rngPbClose     = false   // Scenario 1 closure, Real Pullback against the leg
bool rngClosedNow   = false
int  rngClosedDir   = 0       // direction of the box that closed on this bar
bool rangeSeed      = false

if rangeActive
    // Scenario 2 = Real Breakout of the trend-side boundary:
    //   up-leg box   -> above the ceiling
    //   down-leg box -> below the floor
    bool realBO = rangeDir == 1 ? boUpCeil : boDnFloor
    // Scenario 1 (V15.1): the box closes by a Real Pullback only when the
    // pullback BREAKS the box against its leg:
    //   up-leg box   -> bearish Real Pullback breaking the floor
    //   down-leg box -> bullish Real Pullback breaking the ceiling
    // A pullback pattern completed inside the box is range noise: the box
    // persists. A Real Breakout (Rules 1-3) through the counter-side
    // boundary also closes it.
    bool realPB = rangeDir == 1 ? (pbDnExit or boDnFloor) : (pbUpExit or boUpCeil)

    if realBO
        // ---- SCENARIO 2: Real Breakout closes the Range Box ----
        rngBreakUp    := rangeDir == 1
        rngBreakDn    := rangeDir == -1
        rngClosedNow  := true
        seedBias      := rangeDir
    else if realPB
        // ---- SCENARIO 1: Real Pullback closes the Range Box ----
        rngPbClose   := true
        rngClosedNow := true
        seedBias     := -rangeDir
    else
        // ---- NO CLOSURE: the box persists (see BOUNDARY MECHANICS) ----
        // Ceiling side
        if close > boxTop
            // Close above the ceiling: keep the ceiling so EVERY consecutive pair of
            // candles is tested against it (none is skipped), record the excursion.
            outUpBars += 1
            exHi      := math.max(nz(exHi, high), high)
        else
            // Closed back inside without a Real Breakout / Pullback: the whole
            // excursion (and any wick) belongs to the range -> expand.
            boxTop    := math.max(boxTop, nz(exHi, high), high)
            outUpBars := 0
            exHi      := na
        // Floor side (mirror)
        if close < boxBot
            outDnBars += 1
            exLo      := math.min(nz(exLo, low), low)
        else
            boxBot    := math.min(boxBot, nz(exLo, low), low)
            outDnBars := 0
            exLo      := na

        // Box grows rightward and vertically.
        if not na(rangeBox)
            box.set_right(rangeBox, bar_index)
            box.set_top(rangeBox, boxTop)
            box.set_bottom(rangeBox, boxBot)

    if rngClosedNow
        // ---- CLOSE: freeze as historical boundary, no further expansion ----
        rngClosedDir := rangeDir
        if not na(rangeBox)
            color closeCol = rngPbClose ? rngPbColor : rngBreakUp ? rngBullColor : rngBearColor
            box.set_right(rangeBox, bar_index)
            box.set_top(rangeBox, boxTop)
            box.set_bottom(rangeBox, boxBot)

// MARKET STRUCTURE (CTS / BOS) - implements Market_Structure_Prompt_FA.md
// ---------------------------------------------------------------------
// Up-trend cycle (down-trend is the exact mirror):
//   PHASE A (impulse)  : track H* = highest high of the up impulse. A counter
//                        move WITHOUT one of the 3 Real Pullback conditions is a
//                        range/noise: H* just keeps extending (no CTS, no BOS).
//   -> one of the 3 Real Pullback conditions (bearish) occurs
//   PHASE B (pullback) : H* is LOCKED (pending CTS). Track L* = lowest low of the
//                        WHOLE pullback wave since H*, however long it lasts and
//                        whatever ranges form inside it. Fake moves above H* do
//                        not move it; fake moves below the last BOS do not
//                        reverse the trend.
//   -> Real Breakout (3 conditions) of H*
//        CTS (Recent High) = H*  and  BOS (Key Level) = L*, identified TOGETHER
//        at this bar; a new impulse starts (back to PHASE A).
//   Real Breakout of the last BOS downward (in PHASE A or B) -> trend DOWN.
//        A = H* (highest high of the last up wave) is the unlabelled reversal
//        level until the first bearish BOS is identified.
// Chart start: the trend is undetermined; both directions run the cycle and
// the first complete cycle (impulse -> Real Pullback -> Real Breakout) sets it.
// Every timeframe runs these rules on its own candles.
// =====================================================================
