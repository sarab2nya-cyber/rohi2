// =====================================================================
// DRAWING X-COORDINATE GUARD

atr           = ta.atr(14)

// Range Specific Metrics

// =====================================================================

var float[] maru_sizes = array.new_float(0)

int bigger_count = 0
if array.size(maru_sizes) > 0

        if c_range > array.get(maru_sizes, i)

bool is_big_maru = is_marubozu and (array.size(maru_sizes) < 3 or bigger_count >= 3)

if is_marubozu
    array.push(maru_sizes, c_range)

// =====================================================================
// REAL BREAKOUT (3 Main Conditions) - used by market structure AND the Range Box
// A breakout of `level` is REAL only when it matches one of the three main
// conditions below. Anything else that crosses the level is a Fakeout.
// Condition #4 (PA confirmation look-ahead) is removed everywhere; every rule
// is a closed two-candle pattern evaluated on the bar it completes.
// =====================================================================
f_real_breakout(level, dir) =>
    bool ok = false
    if not na(level)
        if dir == 1
            // Rule 1: two bullish Marubozus, both closing above the level
            bool r1 = is_marubozu[1] and is_marubozu and is_bull[1] and is_bull and (close[1] > level) and (close > level) and (close > close[1])
            float c1_top_body = math.max(close[1], open[1])
            float c1_bot_body = math.min(close[1], open[1])
            float c1_body_tot = math.max(c1_top_body - c1_bot_body, syminfo.mintick)
            float c1_part_above = math.max(0.0, c1_top_body - math.max(c1_bot_body, level))
            bool c1_above_30 = (c1_part_above / c1_body_tot) >= bo_c1_above_pct

            float mid_c1 = (high[1] + low[1]) / 2.0
            // Rule 2: big bullish Marubozu + small bullish candle holding its upper half
            bool r2 = is_big_maru[1] and is_bull[1] and is_bull and (c_range <= bo_c2_max_size * c_range[1]) and (low > mid_c1) and c1_above_30
            // Rule 3: big bullish Marubozu + small bearish candle holding its upper half
            bool r3 = is_big_maru[1] and is_bull[1] and is_bear and (c_range <= bo_c2_max_size * c_range[1]) and (low > mid_c1) and c1_above_30
            ok := r1 or r2 or r3

        else if dir == -1
            bool r1 = is_marubozu[1] and is_marubozu and is_bear[1] and is_bear and (close[1] < level) and (close < level) and (close < close[1])
            float c1_top_body = math.max(close[1], open[1])
            float c1_bot_body = math.min(close[1], open[1])
            float c1_body_tot = math.max(c1_top_body - c1_bot_body, syminfo.mintick)
            float c1_part_below = math.max(0.0, math.min(c1_top_body, level) - c1_bot_body)
            bool c1_below_30 = (c1_part_below / c1_body_tot) >= bo_c1_above_pct

            float mid_c1 = (high[1] + low[1]) / 2.0
            bool r2 = is_big_maru[1] and is_bear[1] and is_bear and (c_range <= bo_c2_max_size * c_range[1]) and (high < mid_c1) and c1_below_30
            bool r3 = is_big_maru[1] and is_bear[1] and is_bull and (c_range <= bo_c2_max_size * c_range[1]) and (high < mid_c1) and c1_below_30
            ok := r1 or r2 or r3
    ok

// =====================================================================
// 100% REAL PULLBACK CONFIRMATION (3 MAIN CONDITIONS)
// =====================================================================
float mid_1 = (nz(high[1], high) + nz(low[1], low)) / 2.0

// BEARISH PULLBACK (Counter-trend down in an Uptrend)
bool pb_r1_dn = is_bear[1] and is_marubozu[1] and is_high_vol[1] and is_bear and is_marubozu and is_high_vol and (c_range >= pb_c2_min_size * nz(c_range[1], c_range)) and (close < nz(close[1], close))
bool pb_r2_dn = is_bear[1] and is_big_maru[1] and is_high_vol[1] and is_bear and (c_range <= pb_c2_max_r2 * nz(c_range[1], c_range)) and (high < mid_1)
bool pb_r3_dn = is_bear[1] and is_big_maru[1] and is_high_vol[1] and is_bull and (c_range <= pb_c2_max_r3 * nz(c_range[1], c_range)) and (high < mid_1)

bool pb_pattern_down = pb_r1_dn or pb_r2_dn or pb_r3_dn

// BULLISH PULLBACK (Counter-trend up in a Downtrend)
bool pb_r1_up = is_bull[1] and is_marubozu[1] and is_high_vol[1] and is_bull and is_marubozu and is_high_vol and (c_range >= pb_c2_min_size * nz(c_range[1], c_range)) and (close > nz(close[1], close))
bool pb_r2_up = is_bull[1] and is_big_maru[1] and is_high_vol[1] and is_bull and (c_range <= pb_c2_max_r2 * nz(c_range[1], c_range)) and (low > mid_1)
bool pb_r3_up = is_bull[1] and is_big_maru[1] and is_high_vol[1] and is_bear and (c_range <= pb_c2_max_r3 * nz(c_range[1], c_range)) and (low > mid_1)

bool pb_pattern_up = pb_r1_up or pb_r2_up or pb_r3_up


var int msTrendState = 0

// =====================================================================
// PRICE ACTION RANGE BOX ENGINE
// ---------------------------------------------------------------------
// State machine:
//   IDLE   --(C1 counter-trend + C2 indecision + no Real Pullback)--> ACTIVE
//   ACTIVE --(trend-side Real Breakout, Rules 1-3)------------------> CLOSED (Scenario 2)
//   ACTIVE --(counter-trend Real Pullback, Rules 1-3, anywhere)----> CLOSED (Scenario 1)
//   ACTIVE --(counter-side Real Breakout, Rules 1-3)----------------> CLOSED (Scenario 1)
//   ACTIVE --(anything else: inside candles, wicks, body whipsaws)--> ACTIVE (expanded by Fakeouts)
//
// BOUNDARY MECHANICS (V13.4) - why the box can no longer swallow a trend:
//   * The box is never closed by "time" or by a partial confirmation;
//     only Rules 1-3 close it.
//   * WICK SWEEP: a candle that pokes beyond a boundary but CLOSES inside
//     is internal liquidity -> the boundary absorbs its wick at once.
//   * BODY BREAK: a candle that CLOSES beyond a boundary is a break
//     attempt. The boundary is NOT moved while closes stay beyond it, so
//     Rules 1-3 are always tested against the real ceiling/floor. (V13.3
//     moved the boundary to every new extreme, so a trend kept raising the
//     bar it had to break and the box grew with the whole trend.)
//   * FAKEOUT: the attempt fails when price closes back inside. If that
//     happens within `Max Fakeout Bars` closes, the excursion is a Fakeout
//     and the box expands to contain it.
//   * A longer excursion that later comes back was a move away from the
//     range, not internal liquidity: it is NOT absorbed (V13.3 absorbed a
//     100-bar trend leg on its first close back inside, inflating the box
//     many times over).
//
// While ACTIVE every candle is internal liquidity: no pullback is
// confirmed, no CTS/BOS is issued, no signal dot is plotted. Only the
// candle that closes the box may carry a signal.
// =====================================================================
var bool  rangeActive   = false
var int   rangeDir      = 0      // +1 box in an up leg (C1 bearish), -1 box in a down leg (C1 bullish)
var float boxTop        = na     // ceiling: C1/C2 extremes + absorbed wick sweeps and Fakeouts
var float boxBot        = na     // floor:   C1/C2 extremes + absorbed wick sweeps and Fakeouts
var int   outUpBars     = 0      // consecutive closes above the ceiling (current break attempt)
var int   outDnBars     = 0      // consecutive closes below the floor   (current break attempt)
var float exHi          = na     // highest high of the current upside excursion
var float exLo          = na     // lowest low of the current downside excursion
var int   rangeLeftBar  = na     // bar of Candle 1 (the anchor candle, always retained)
var box   rangeBox      = na
var int   lastCloseBar  = na     // bar on which the last box closed
var int   seedBias      = 0      // direction of the last real move (+1 up / -1 down), 0 = use msTrendState

// Closed Range registry (historical boundaries). A structural level whose
// price is strictly inside a closed box and whose bar lies inside its span
// can never become CTS / BOS / Key Level.
var float[] crTop   = array.new_float()
var float[] crBot   = array.new_float()
var int[]   crLeft  = array.new_int()
var int[]   crRight = array.new_int()

f_inClosedRange(float price, int barIdx) =>
    bool inside = false
    if not na(price) and not na(barIdx) and array.size(crTop) > 0
        for i = 0 to array.size(crTop) - 1
            if barIdx >= array.get(crLeft, i) and barIdx <= array.get(crRight, i) and price < array.get(crTop, i) and price > array.get(crBot, i)
                inside := true
                break
    inside

// Closure tests against the current box boundaries (which never move while
// closes are beyond them). Evaluated on every bar (not inside a
// conditional) so the history-referencing functions stay consistent.
float ceilLvl  = rangeActive ? boxTop : na
float floorLvl = rangeActive ? boxBot : na
bool boUpCeil  = f_real_breakout(ceilLvl, 1)        // Real Breakout above the ceiling
bool boDnFloor = f_real_breakout(floorLvl, -1)      // Real Breakout below the floor

// ---------------- Candle 1 (Trigger): counter-trend candle ----------------
float c1Range   = bar_index >= 1 ? math.max(high[1] - low[1], syminfo.mintick) : syminfo.mintick
float c1Body    = bar_index >= 1 ? math.abs(close[1] - open[1]) : 0
float c1BodyPct = c1Body / c1Range
float c1Upper   = bar_index >= 1 ? high[1] - math.max(open[1], close[1]) : 0
float c1Lower   = bar_index >= 1 ? math.min(open[1], close[1]) - low[1] : 0

bool c1Bear       = bar_index >= 1 and close[1] < open[1]
bool c1Bull       = bar_index >= 1 and close[1] > open[1]
bool c1ValidBody  = c1Body > syminfo.mintick
bool c1Doji       = c1BodyPct <= rngDojiPct
bool c1Maru       = c1ValidBody and c1BodyPct > bo_maru_body
bool c1Normal     = c1ValidBody and c1BodyPct >= rngNormalMinPct and c1BodyPct <= rngNormalMaxPct
bool c1UpperPin   = c1ValidBody and c1Upper >= c1Body * rngPinWickRatio and c1Upper > c1Lower
bool c1LowerPin   = c1ValidBody and c1Lower >= c1Body * rngPinWickRatio and c1Lower > c1Upper
bool c1Pin        = c1UpperPin or c1LowerPin
bool c1FormValid  = c1Pin or c1Doji or c1Normal or c1Maru   // the 4 permitted forms

// ---------------- Candle 2 (Confirmation): indecision test ----------------
// Protrusion = part of C2's total height (body + wicks) outside C1's range.
bool  c2Inside     = bar_index >= 1 and high <= high[1] and low >= low[1]
float c2Protrude   = bar_index >= 1 ? math.max(high - high[1], 0) + math.max(low[1] - low, 0) : c_range
bool  c2Within30   = (c2Protrude / c_range) <= rngMaxOutsidePct

// C2 is valid if it satisfies ONE of the four conditions:
//  A) complete Inside Bar (body and wicks inside C1's range)
//  B) max 30% protrusion (for non-Normal / non-Marubozu candles; Normal
//     candles are governed by D, momentum Marubozus only qualify via A)
//  C) Pin Bar or Doji
//  D) Normal (50-70% body) ONLY if C1 is Pin Bar/Doji AND the 30% rule is met
bool c2CondA = c2Inside
bool c2CondB = c2Within30 and not rngNormal and not rngMaru
bool c2CondC = rngPinBar or rngDoji
bool c2CondD = rngNormal and (c1Pin or c1Doji) and c2Within30
bool c2Valid = bar_index >= 1 and (c2CondA or c2CondB or c2CondC or c2CondD)

// ---------------- Pullback check before formation ----------------
// If C1 (bar -1) or C2 (bar 0) completes a Real Pullback, it is a
// pullback, NOT a Range Box.
bool noPullbackUp = not pb_pattern_down and not pb_pattern_down[1]  // up leg: no bearish Real Pullback
bool noPullbackDn = not pb_pattern_up   and not pb_pattern_up[1]    // down leg: no bullish Real Pullback

// ---------------- Range initiation after a breakout ----------------
// "Counter-trend" is measured against the LAST REAL MOVE: after a box is
// broken downward the next box needs a bullish C1, after an upward break
// a bearish C1. Before any box has closed, the market-structure trend is
// used. C1 must be a candle AFTER the closing candle (the closing candle
// is never reused as C1), and the box only opens if C2 also qualifies.
int  legDir      = seedBias != 0 ? seedBias : msTrendState
bool c1AfterExit = na(lastCloseBar) or (bar_index - 1) > lastCloseBar

bool seedUp = legDir == 1  and c1AfterExit and c1Bear and c1FormValid and c2Valid and noPullbackUp
bool seedDn = legDir == -1 and c1AfterExit and c1Bull and c1FormValid and c2Valid and noPullbackDn

// ---------------- Per-bar Range events ----------------
bool rangeWasActive = rangeActive
bool rngBreakUp     = false   // Scenario 2 closure, up-leg box broken above its ceiling
bool rngBreakDn     = false   // Scenario 2 closure, down-leg box broken below its floor
bool rngPbClose     = false   // Scenario 1 closure, Real Pullback against the leg
bool rngClosedNow   = false
int  rngClosedDir   = 0       // direction of the box that closed on this bar
bool rangeSeed      = false

if rangeActive
    // Scenario 2 = Real Breakout of the trend-side boundary:
    //   up-leg box   -> above the ceiling
    //   down-leg box -> below the floor
    bool realBO = rangeDir == 1 ? boUpCeil : boDnFloor
    // Scenario 1 (V13.5): a valid Real Pullback (Rules 1-3) against the leg
    // closes the box ON THE CANDLE THAT COMPLETES IT, wherever that candle
    // is (inside or outside the box):
    //   up-leg box   -> bearish Real Pullback
    //   down-leg box -> bullish Real Pullback
    // A Real Breakout (Rules 1-3) through the counter-side boundary also
    // closes it: a breakout against the leg is at least a pullback of it,
    // and without it a down-leg box could never be closed by a rally.
    bool realPB = rangeDir == 1 ? (pb_pattern_down or boDnFloor) : (pb_pattern_up or boUpCeil)

    if realBO
        // ---- SCENARIO 2: Real Breakout closes the Range Box ----
        rngBreakUp    := rangeDir == 1
        rngBreakDn    := rangeDir == -1
        rngClosedNow  := true
        seedBias      := rangeDir
    else if realPB
        // ---- SCENARIO 1: Real Pullback closes the Range Box ----
        rngPbClose   := true
        rngClosedNow := true
        seedBias     := -rangeDir
    else
        // ---- NO CLOSURE: the box persists (see BOUNDARY MECHANICS) ----
        // Ceiling side
        if close > boxTop
            // Body break attempt: keep the ceiling, record the excursion.
            outUpBars += 1
            exHi      := math.max(nz(exHi, high), high)
        else
            // Closed back inside: sweep (0 closes outside) or Fakeout
            // (<= Max Fakeout Bars) -> absorb; longer excursion -> ignore.
            if outUpBars <= rngMaxFakeoutBars
                boxTop := math.max(boxTop, nz(exHi, high), high)
            outUpBars := 0
            exHi      := na
        // Floor side (mirror)
        if close < boxBot
            outDnBars += 1
            exLo      := math.min(nz(exLo, low), low)
        else
            if outDnBars <= rngMaxFakeoutBars
                boxBot := math.min(boxBot, nz(exLo, low), low)
            outDnBars := 0
            exLo      := na

        // Box grows rightward and vertically.
        if not na(rangeBox)
            box.set_right(rangeBox, bar_index)
            box.set_top(rangeBox, boxTop)
            box.set_bottom(rangeBox, boxBot)

    if rngClosedNow
        // ---- CLOSE: freeze as historical boundary, no further expansion ----
        rngClosedDir := rangeDir
        if not na(rangeBox)
            color closeCol = rngPbClose ? rngPbColor : rngBreakUp ? rngBullColor : rngBearColor
            box.set_right(rangeBox, bar_index)
            box.set_top(rangeBox, boxTop)
            box.set_bottom(rangeBox, boxBot)
            box.set_bgcolor(rangeBox, closeCol)
            box.set_border_color(rangeBox, color.new(closeCol, 15))
            box.set_border_style(rangeBox, line.style_solid)
            box.set_text(rangeBox, rngPbClose ? "Range: Real Pullback" : "Range: Real Breakout")
            box.set_text_size(rangeBox, size.tiny)
            box.set_text_color(rangeBox, color.new(closeCol, 0))
            box.set_text_halign(rangeBox, text.align_left)
            box.set_text_valign(rangeBox, (rngBreakDn or (rngPbClose and rangeDir == 1)) ? text.align_bottom : text.align_top)

        // Register the interior (the closing candle itself is excluded: it
        // is the one candle allowed to carry a signal).
        array.push(crTop, boxTop)
        array.push(crBot, boxBot)
        array.push(crLeft, rangeLeftBar)
        array.push(crRight, bar_index - 1)
        if array.size(crTop) > rngMemory
            array.shift(crTop)
            array.shift(crBot)
            array.shift(crLeft)
            array.shift(crRight)

        rangeActive  := false
        rangeDir     := 0
        outUpBars    := 0
        outDnBars    := 0
        exHi         := na
        exLo         := na
        lastCloseBar := bar_index
        rangeBox     := na

// ---- FORMATION: C1 + C2 valid and no Real Pullback -> box becomes ACTIVE ----
// The box STARTS ON CANDLE 1 (the first counter-trend candle) and its
// initial boundaries are Candle 1's own High and Low (V13.5). Candle 2 only
// confirms the indecision; it does not widen the box. Any later wick
// sweep / Fakeout beyond C1's range is absorbed by the normal expansion.
if not rangeWasActive and not rngClosedNow and (seedUp or seedDn)
    rangeSeed    := true
